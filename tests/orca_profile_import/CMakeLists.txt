cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(OrcaProfileImportTests LANGUAGES CXX)
    enable_testing()
endif()
set(IMPORT_DIR "${CMAKE_CURRENT_LIST_DIR}/../../src/slic3r-shared/src/Slic3r/Biz/Preset/IO")
add_executable(orca_profile_import_tests tests.cpp "${IMPORT_DIR}/OrcaProfileConverter.cpp")
target_include_directories(orca_profile_import_tests PRIVATE "${IMPORT_DIR}")
if(TARGET nlohmann_json::nlohmann_json)
    target_link_libraries(orca_profile_import_tests PRIVATE nlohmann_json::nlohmann_json)
else()
    find_path(JSON_INCLUDE_DIR nlohmann/json.hpp REQUIRED)
    target_include_directories(orca_profile_import_tests PRIVATE "${JSON_INCLUDE_DIR}")
endif()
target_compile_features(orca_profile_import_tests PRIVATE cxx_std_20)
target_compile_definitions(orca_profile_import_tests PRIVATE JSON_USE_IMPLICIT_CONVERSIONS=0)
set_property(TARGET orca_profile_import_tests PROPERTY DISABLE_PRECOMPILE_HEADERS ON)
add_test(NAME orca_profile_import COMMAND orca_profile_import_tests)
if(TARGET slic3r-shared AND TARGET slic3r-biz-parser)
    find_package(nlohmann_json 3.11 REQUIRED)
    add_executable(orca_profile_native_tests native_tests.cpp)
    target_link_libraries(orca_profile_native_tests PRIVATE slic3r-shared slic3r-biz-parser nlohmann_json::nlohmann_json)
    set_property(TARGET orca_profile_native_tests PROPERTY DISABLE_PRECOMPILE_HEADERS ON)
    add_test(NAME orca_profile_native COMMAND orca_profile_native_tests)
endif()
