cmake_minimum_required(VERSION 3.13)
project(camera_video_smoke LANGUAGES CXX)
if(APPLE)
    enable_language(OBJCXX)
endif()
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../cmake/modules")
find_package(FFmpeg REQUIRED)
find_package(Threads REQUIRED)
add_executable(video_stream_decoder_smoke
    video_stream_decoder_smoke.cpp
    ../../src/slic3r/Utils/VideoStreamDecoder.cpp)
if(APPLE)
    target_sources(video_stream_decoder_smoke PRIVATE ../../src/slic3r/Utils/CameraDevicesMac.mm)
else()
    target_sources(video_stream_decoder_smoke PRIVATE ../../src/slic3r/Utils/CameraDevices.cpp)
endif()
target_compile_features(video_stream_decoder_smoke PRIVATE cxx_std_17)
target_include_directories(video_stream_decoder_smoke PRIVATE ../../src)
target_link_libraries(video_stream_decoder_smoke PRIVATE Threads::Threads FFmpeg::FFmpeg)

set(VIDEO_TEST_URI "" CACHE STRING "Video URI for the embedded decoder integration test")
if(VIDEO_TEST_URI)
    enable_testing()
    add_test(NAME video_stream_decoder COMMAND video_stream_decoder_smoke "${VIDEO_TEST_URI}")
    set_tests_properties(video_stream_decoder PROPERTIES TIMEOUT 60)
endif()
